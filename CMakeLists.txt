cmake_minimum_required(VERSION 3.20)
project(CHIP8Studio VERSION 3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(CHIP8_SANITIZERS "Address/undefined sanitizers (GCC/Clang core tests)" OFF)
add_library(chip8_core src/Chip8Core.cpp)
target_include_directories(chip8_core PUBLIC src)
if(MSVC)
  target_compile_options(chip8_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(chip8_core PRIVATE -Wall -Wextra -Wpedantic)
  if(CHIP8_SANITIZERS)
    target_compile_options(chip8_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(chip8_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(CHIP8.Tests tests/Chip8Tests.cpp)
target_link_libraries(CHIP8.Tests PRIVATE chip8_core)
add_executable(CHIP8.RomRunner tests/RomRunner.cpp)
target_link_libraries(CHIP8.RomRunner PRIVATE chip8_core)
add_executable(CHIP8.MediaFixture tests/MediaFixture.cpp)
target_include_directories(CHIP8.MediaFixture PRIVATE src)
add_executable(CHIP8.CatalogRunner tests/CatalogRunner.cpp)
target_link_libraries(CHIP8.CatalogRunner PRIVATE chip8_core)
if(MSVC)
  target_compile_options(CHIP8.Tests PRIVATE /W4 /utf-8)
  target_compile_options(CHIP8.RomRunner PRIVATE /W4 /utf-8)
  target_compile_options(CHIP8.MediaFixture PRIVATE /W4 /utf-8)
  target_compile_options(CHIP8.CatalogRunner PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_test(NAME core COMMAND CHIP8.Tests)
add_test(NAME catalog_boot COMMAND CHIP8.CatalogRunner "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/catalog-boot.json")
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
  add_test(NAME gif_media COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_media.py" "$<TARGET_FILE:CHIP8.MediaFixture>")
  add_test(NAME release_publication COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_release_publish.py")
  add_test(NAME timendus COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_roms.py" --runner "$<TARGET_FILE:CHIP8.RomRunner>" --output "${CMAKE_CURRENT_BINARY_DIR}/rom-results")
endif()
if(WIN32)
  enable_language(RC)
  add_executable(CHIP8.Emulator WIN32 src/WindowsMain.cpp src/resource.rc)
  target_compile_definitions(CHIP8.Emulator PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(CHIP8.Emulator PRIVATE chip8_core comdlg32 shell32 winmm dwmapi gdiplus ole32 bcrypt comctl32 mfuuid)
  if(MSVC)
    target_compile_options(CHIP8.Emulator PRIVATE /W4 /utf-8)
    target_link_options(CHIP8.Emulator PRIVATE /MANIFEST:NO)
  else()
    target_link_options(CHIP8.Emulator PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(CHIP8.Tests PRIVATE -static)
    target_link_options(CHIP8.RomRunner PRIVATE -static)
    target_link_options(CHIP8.MediaFixture PRIVATE -static)
    target_link_options(CHIP8.CatalogRunner PRIVATE -static)
  endif()
endif()
